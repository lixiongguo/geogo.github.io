```
初始化策略Pi(s) = Uniform in {a1,a2,a3,....}
Repeat(多轮迭代)
	for (s,r,a) in episode:
		G = r + gamma * G 
		Q(s,a) = Q(s) + l*(G - Q(s,a))
	策略提升Pi(s) = Onehot(index == argmaxQ(s,a))
```

\begin{algorithm}
\caption{Monte Carlo evaluate Q(s,a)}
\begin{algorithmic}
\State \textbf{Input:} Initial stochastic policy \pi()
\State \textbf{Output:} Optimal stochastic policy \pi^*()
\State Initialize an empty set for the MST: \( T \gets \emptyset \)
\State Sort all edges in \( E \) in non-decreasing order of their weights: \( E \gets \text{Sort}(E) \)
\State Initialize a disjoint-set data structure for all vertices: \( \text{Make-Set}(v) \) for each \( v \in V \)

\For{each edge \( (u, v) \) in \( E \) (in sorted order)}
    \State \textbf{if} \( \text{Find-Set}(u) \neq \text{Find-Set}(v) \)
        \State \( T \gets T \cup \{(u, v)\} \)
        \State \text{Union}(\text{Find-Set}(u), \text{Find-Set}(v))
\EndFor

\State \textbf{return} \( T \)
\end{algorithmic}
\end{algorithm}
